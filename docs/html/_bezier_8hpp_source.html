30 template<typename T>
31 inline T QuadraticInterpolate(T const& p1, T const& p2, T const& p3, float t) noexcept
32 {
33 return p1 * (1 - t) * (1 - t) +
34 p2 * 2 * t * (1 - t) +
35 p3 * t * t;
36 }

44 template<>
45 inline glm::vec3 QuadraticInterpolate(glm::vec3 const& p1, glm::vec3 const& p2, glm::vec3 const& p3, float t) noexcept
46 {
47 const glm::vec3 t0 = glm::vec3((1 - t) * (1 - t));
48 const glm::vec3 t1 = glm::vec3(2 * t * (1 - t));
49 const glm::vec3 t2 = glm::vec3(t * t);
50
51 glm::vec3 result = p1 * t0;
52 result = p2 * t1 + result;
53 result = p3 * t2 + result;
54
55 return result;
56 }

68 template<typename T>
69 inline T CubicInterpolate(T const& p1, T const& p2, T const& p3, T const& p4, float t) noexcept
70 {
71 return p1 * (1 - t) * (1 - t) * (1 - t) +
72 p2 * 3 * t * (1 - t) * (1 - t) +
73 p3 * 3 * t * t * (1 - t) +
74 p4 * t * t * t;
75 }

84 template<>
85 inline glm::vec3 CubicInterpolate(glm::vec3 const& p1, glm::vec3 const& p2, glm::vec3 const& p3, glm::vec3 const& p4, float t) noexcept
86 {
87 const glm::vec3 t0 = glm::vec3((1 - t) * (1 - t) * (1 - t));
88 const glm::vec3 t1 = glm::vec3(3 * t * (1 - t) * (1 - t));
89 const glm::vec3 t2 = glm::vec3(3 * t * t * (1 - t));
90 const glm::vec3 t3 = glm::vec3(t * t * t);

93 result = p2 * t1 + result;
94 result = p3 * t2 + result;
95 result = p4 * t3 + result;

109 template<typename T>
110 inline T CubicTangent(T const& p1, T const& p2, T const& p3, T const& p4, float t) noexcept
111 {
112 return p1 * (-1 + 2 * t - t * t) +
113 p2 * (1 - 4 * t + 3 * t * t) +
114 p3 * (2 * t - 3 * t * t) +
115 p4 * (t * t);
116 }

150 template<typename TOutputFunc>
151 void CreatePatchVertices(_In_reads_(16) const glm::vec3 patch[16], std::size_t tessellation, bool isMirrored, TOutputFunc outputVertex)
152 {
153 for (std::size_t i = 0; i <= tessellation; i++)
154 {
155 const float u = float(i) / float(tessellation);
156
157 for (std::size_t j = 0; j <= tessellation; j++)
158 {
159 const float v = float(j) / float(tessellation);
160
161 // Perform four horizontal bezier interpolations
162 // between the control points of this patch.
163 const glm::vec3 p1 = CubicInterpolate(patch[0], patch[1], patch[2], patch[3], u);
164 const glm::vec3 p2 = CubicInterpolate(patch[4], patch[5], patch[6], patch[7], u);
165 const glm::vec3 p3 = CubicInterpolate(patch[8], patch[9], patch[10], patch[11], u);
166 const glm::vec3 p4 = CubicInterpolate(patch[12], patch[13], patch[14], patch[15], u);
167
168 // Perform a vertical interpolation between the results of the
169 // previous horizontal interpolations, to compute the position.
170 const glm::vec3 position = CubicInterpolate(p1, p2, p3, p4, v);
171
172 // Perform another four bezier interpolations between the control
173 // points, but this time vertically rather than horizontally.
174 const glm::vec3 q1 = CubicInterpolate(patch[0], patch[4], patch[8], patch[12], v);
175 const glm::vec3 q2 = CubicInterpolate(patch[1], patch[5], patch[9], patch[13], v);
176 const glm::vec3 q3 = CubicInterpolate(patch[2], patch[6], patch[10], patch[14], v);
177 const glm::vec3 q4 = CubicInterpolate(patch[3], patch[7], patch[11], patch[15], v);
178
179 // Compute vertical and horizontal tangent vectors.
180 const glm::vec3 tangent1 = CubicTangent(p1, p2, p3, p4, v);
181 const glm::vec3 tangent2 = CubicTangent(q1, q2, q3, q4, u);
182
183 // Cross the two tangent vectors to compute the normal.
184 glm::vec3 normal = glm::cross(tangent1, tangent2);
185
186 if (glm::length(normal) > std::numeric_limits<float>::epsilon())
187 {
188 normal = glm::normalize(normal);
189
190 // If this patch is mirrored, we must invert the normal.
191 if (isMirrored)
192 {
193 normal = -normal;
194 }
195 }
196 else
197 {
198 // In a tidy and well constructed bezier patch, the preceding
199 // normal computation will always work. But the classic teapot
200 // model is not tidy or well constructed! At the top and bottom
201 // of the teapot, it contains degenerate geometry where a patch
202 // has several control points in the same place, which causes
203 // the tangent computation to fail and produce a zero normal.
204 // We 'fix' these cases by just hard-coding a normal that points
205 // either straight up or straight down, depending on whether we
206 // are on the top or bottom of the teapot. This is not a robust
207 // solution for all possible degenerate bezier patches, but hey,
208 // it's good enough to make the teapot work correctly!
209
210 normal = (position.y < 0) ? glm::vec3(0, -1, 0) : glm::vec3(0, 1, 0);
211 }
212
213 // Compute the texture coordinate.
214 const float mirroredU = isMirrored ? 1 - u : u;
215
216 const glm::vec3 textureCoordinate = glm::vec3(mirroredU, v, 0);
217
218 // Output this vertex.
219 outputVertex(position, normal, textureCoordinate);
220 }
221 }
222 }

231 template<typename TOutputFunc>
232 void CreatePatchIndices(std::size_t tessellation, bool isMirrored, TOutputFunc outputIndex)
233 {
234 std::size_t stride = tessellation + 1;
235
236 for (std::size_t i = 0; i < tessellation; i++)
237 {
238 for (std::size_t j = 0; j < tessellation; j++)
239 {
240 // Make a list of six index values (two triangles).
241 std::array<std::size_t, 6> indices =
242 {
243 i * stride + j,
244 (i + 1) * stride + j,
245 (i + 1) * stride + j + 1,
246
247 i * stride + j,
248 (i + 1) * stride + j + 1,
249 i * stride + j + 1,
250 };
251
252 // If this patch is mirrored, reverse indices to fix the winding order.
253 if (isMirrored)
254 {
255 std::reverse(indices.begin(), indices.end());
256 }
257
258 // Output these index values.
259 std::for_each(indices.begin(), indices.end(), outputIndex);
260 }
261 }
262 }
