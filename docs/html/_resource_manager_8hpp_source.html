110 template<typename T>
111 [[nodiscard]] ResourceHandle<T> LoadResource(GUID p_guid, std::function<void(ResourceHandle<T>)> p_onLoadCallback = nullptr)
112 {
113 static_assert(std::is_base_of_v<IResource, T>, "T must inherit from IResource");
114 assert(m_isInitialized && "ResourceManager is not initialized.");
115
116 MetaEntry metaEntry;
117 if (!m_registry.GetResourceMetaData(p_guid, metaEntry))
118 {
119 // Handle missing resource
120 // TODO: Log this as a warning/error
121 return ResourceHandle<T>(C_INVALID_GUID, this);
122 }
123
124 bool loadAsync = false;
125 auto [it, wasInserted] = m_liveResources.try_emplace(p_guid);
126 if (wasInserted)
127 {
128 // Resource was just created in-place -> Update its state
129 it->second.state = ResourceState::LoadingAsync;
130 it->second.refCount = 0; // Incremented when handle is constructed
131 loadAsync = true;
132 }
133
134 ResourceHandle<T> handle(p_guid, this);
135
136 // Handle callback
137 if (p_onLoadCallback)
138 {
139 if (it->second.state == ResourceState::Ready)
140 {
141 p_onLoadCallback(handle);
142 }
143 else if (it->second.state != ResourceState::Failed)
144 {
145 // Resource is not loaded and has not failed -> Store callback until loaded
146 // Capture handle by value inside lambda to guarantee that ref count is >= 1
147 it->second.loadCallbacks.push_back([p_onLoadCallback, handle]()
148 {
149 p_onLoadCallback(handle);
150 });
151 }
152 }
153
154 if (loadAsync)
155 {
156 PushWorkerLoadTask<T>(p_guid, metaEntry);
157 }
158
159 return handle;
160 }

238 template<typename T>
239 T *GetResource(GUID p_guid)
240 {
241 assert(m_isInitialized && "AssetManager is not initialized.");
242
243 auto it = m_liveResources.find(p_guid);
244 if (it != m_liveResources.end() && it->second.state == ResourceState::Ready)
245 {
246 // Asset is ready to be used
247 return static_cast<T*>(it->second.resource.get());
248 }
249
250 // Resource with guid was not found or is not ready -> hand over fallback for that resource type
251 auto fallbackIt = m_fallbackResources.find(typeid(T));
252 if (fallbackIt != m_fallbackResources.end())
253 {
254 return static_cast<T*>(fallbackIt->second.get());
255 }
256
257 return nullptr; // Worst case scenario return nullptr
258 }

281 template<typename T>
282 void PushWorkerLoadTask(GUID p_guid, MetaEntry p_metaEntry)
283 {
284 ThreadPool::GetInstance().PushTask([this, p_guid, p_metaEntry]()
285 {
286 std::filesystem::path absAssetPath = std::filesystem::absolute(m_rootDirectory) / p_metaEntry.relAssetPath;
287 std::unique_ptr<T> loadedResource = ResourceTraits<T>::LoadCPU(absAssetPath, p_metaEntry.sourceName, p_metaEntry.loadSettings);
288
289 AsyncLoadResult res;
290 res.succeeded = (loadedResource != nullptr);
291 res.guid = p_guid;
292 res.resource = loadedResource.release();
293 res.loadFlags = p_metaEntry.loadFlags;
294
295 m_asyncLoadResults.Push(res);
296 });
297 }
